93、

通过对检材“02-crime\_records.sql”分析，统计相邻两次作案时间间隔在1天之内的城市和该城市两次作案时间间隔在1天之内的案件总数量，找出案件总数最多的城市名。（答案格式：按照实际情况填写）（ ）

这个文件需要sql8.0进行打开导入

![image](images/AitQ-4-6sQKud8J66xld3WwFHHnD-WDq_7c6mLv3U8A.png)

于是切换版本

![image](images/Xk8qDVciL3ACcEhAQLWKuQUoEIbQzeGaJwhesv0FXEk.png)

```sql
-- 步骤1：为每个城市的案件按时间排序，匹配上一次作案记录并计算时间间隔
WITH crime_with_prev AS (
    SELECT
        city,  -- 城市字段（需确认实际字段名）
        crime_time,  -- 作案时间字段（需确认实际字段名）
        -- 获取同城市上一次作案的时间（按时间排序）
        LAG(crime_time) OVER (PARTITION BY city ORDER BY crime_time) AS prev_crime_time
    FROM
        crime_records  -- 案件表名（需替换为实际表名）
    WHERE
        crime_time IS NOT NULL  -- 排除无作案时间的无效记录
),
-- 步骤2：筛选出相邻间隔≤1天的记录，并按城市统计数量
city_crime_count AS (
    SELECT
        city,
        COUNT(*) AS interval_1day_count  -- 相邻间隔≤1天的案件总数
    FROM
        crime_with_prev
    WHERE
        prev_crime_time IS NOT NULL  -- 排除每个城市的第一条案件（无上次记录）
        -- 计算时间间隔（天），≤1天则符合条件
        AND DATEDIFF(crime_time, prev_crime_time) <= 1
    GROUP BY
        city
)
-- 步骤3：找出案件总数最多的城市
SELECT
    city AS max_crime_city,  -- 案件数最多的城市名
    interval_1day_count AS max_count  -- 该城市的符合条件案件总数
FROM
    city_crime_count
ORDER BY
    interval_1day_count DESC
LIMIT 1;
```
